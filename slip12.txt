Indoor.java
package game;

public class Indoor 
{
    String gameName;
    int players;
    public Indoor() 
    {
        gameName = "Unknown";
        players = 0;
    }
    public Indoor(String name, int p)
    {
        gameName = name;
        players = p;
    }
    public void display()
    {
        System.out.println("Indoor Game: " + gameName);
        System.out.println("Number of Players: " + players);
    }
}
