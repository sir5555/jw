#JAVA
import java.util.*;
class ArrayProgram
{
    public static void main(String args[])
    {
        Scanner sc = new Scanner(System.in);
        int a[] = new int[5];
        int sum = 0;
        int i, j, temp;
        System.out.println("Enter 5 elements:");
        for(i = 0; i < 5; i++)
        {
            a[i] = sc.nextInt();
            sum = sum + a[i];
        }
        System.out.println("Sum = " + sum);
        for(i = 0; i < 5; i++)
        {
            for(j = i + 1; j < 5; j++)
            {
                if(a[i] > a[j])
                {
                    temp = a[i];
                    a[i] = a[j];
                    a[j] = temp;
                }
            }
        }
        System.out.println("Array in Ascending Order:");
        for(i = 0; i < 5; i++)
            System.out.print(a[i] + " ");
    }
}
